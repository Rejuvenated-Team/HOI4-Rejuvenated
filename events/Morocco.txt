#Independance
country_event = {
	id = morocco.100
	title = morocco.100.t
	desc = morocco.100.d
	picture = GFX_report_event_election_vote
	
	is_triggered_only = yes
	
	option = { # Accept demands
		name = morocco.100.a
		ai_chance = {
			base = 25 
			modifier = {
				factor = 50
				has_war = yes
				any_enemy_country = {
					strength_ratio = { 
						tag = ROOT 
						ratio < 0.4
					}
				}
			}
			modifier = {
				factor = 75
				16 = { NOT = { is_controlled_by = FRA } } #Paris	
			}
		} 
	}
	option = { # Refuse
		name = morocco.100.b
		ai_chance = { base = 25 }
		FRA = {
			create_wargoal = {
				type = annex_everything
				target = MOR
			}
		}
	}		
}
