	id = RJ_NZL.4a

	desc = RJ_NZL.4d_a

		name = "Delightful! Let's make it official."
		ai_chance = {
			base = 1
		}
		if = {
			limit = {
				JAP = {
					NOT = {
						is_in_faction = yes
					}
				}
			}
			create_faction_from_template = faction_template_pacific_alliance_faction
			add_to_faction = JAP
		}
		else_if = {
			limit = {
				JAP = {
					is_in_faction = yes
				}
			}
			JAP = {
				add_to_faction = NZL
			}
		}
		add_to_tech_sharing_group = japanese_research
		if = {
			limit = {
				JAP = {
					NOT = {
						is_in_tech_sharing_group = japanese_research
					}
				}
			}
			JAP = {
				add_to_tech_sharing_group = japanese_research
			}
		}
		add_ideas = RJ_NZL_idea_new_zealand_likes_japan
		JAP = {
			add_ideas = RJ_NZL_idea_japan_likes_new_zealand
		}
	}
}
