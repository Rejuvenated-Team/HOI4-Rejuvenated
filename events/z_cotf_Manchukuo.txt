country_event = {
	id = cotf_man.1
	title = "[FROM.GetLeader] Proposes Unification"
	desc = cotf_man.1.d
	picture = GFX_report_event_china_politicians_captured

	is_triggered_only = yes
	
	#sure why not
	option = {
		name = "The Nationalists will not be able to defeat our great union!"
		ai_chance = { 
			base = 100
		}
		if = {
			limit = {
				ROOT = {
					is_ai = yes
				}
				MAN = {
					is_ai = no
				}
			}
			ROOT = {
				change_tag_from = MAN
			}
		}
		if = {
			limit = {
				KOR = {
					exists = yes
					is_subject_of = MAN
				}
			}
			MAN = {
				set_autonomy = {
					target = KOR
					autonomy_state = autonomy_free
					end_wars = no
					end_civil_wars = no
				}
			}
			ROOT = {
				puppet = KOR
			}
		}
		MAN = {
			every_unit_leader = {
				set_nationality = ROOT
			}
		}
		ROOT = {
			annex_country = {
				target = MAN
				transfer_troops = yes
			}
		}
	}

	#nah
	option = {
		name = "We don't have time for this."
		ai_chance = { 
			base = 0
		}
	}
}
